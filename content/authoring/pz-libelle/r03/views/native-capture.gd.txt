extends SceneTree
const Session = preload("res://model/session.gd")
const Board = preload("res://.libelle_probe_board.gd")
var output: String = "E:/Zeuch/Coding/picross/.codex-libelle-step3-20261115/source/views"
func _initialize() -> void:
	call_deferred("run")
func run() -> void:
	var raw: Dictionary = JSON.parse_string(FileAccess.get_file_as_string("E:/Zeuch/Coding/picross/.codex-libelle-step3-20261115/source/candidate-definition.json"))
	var session = Session.new(raw)
	for y: int in range(15):
		for x: int in range(20):
			session.player.cells[y*20+x] = int(raw.solution[y][x]) if int(raw.solution[y][x]) > 0 else -1
	var surface := SubViewport.new()
	surface.render_target_update_mode = SubViewport.UPDATE_ALWAYS
	root.add_child(surface)
	var paper := TextureRect.new()
	paper.texture = load("res://book/paper.png")
	paper.expand_mode = TextureRect.EXPAND_IGNORE_SIZE
	paper.stretch_mode = TextureRect.STRETCH_SCALE
	var mat := ShaderMaterial.new()
	var shader := Shader.new()
	shader.code = "shader_type canvas_item; render_mode unshaded; void fragment() { COLOR = vec4(mix(texture(TEXTURE, UV*0.72+vec2(0.11,0.17)).rgb,vec3(1.0,0.9804,0.9412),0.55),1.0); }"
	mat.shader = shader
	paper.material = mat
	surface.add_child(paper)
	var board := Board.new()
	board.session = session
	board.book_layout = true
	board.mode = "V"
	board.animations = false
	surface.add_child(board)
	var metadata: Array = []
	for cell: int in [24,16]:
		surface.size = Vector2i(cell*20+160,cell*15+130)
		paper.size = Vector2(surface.size)
		board.size = Vector2(surface.size)
		board.requested_cell = cell
		board.layout_key = ""
		board._layout()
		for status: int in range(3):
			board.probe_status = status
			board.queue_redraw()
			board.marks.sync()
			await process_frame
			await process_frame
			await RenderingServer.frame_post_draw
			var file: String = "native-%d-%d.png" % [cell,status]
			surface.get_texture().get_image().save_png(output.path_join(file))
			metadata.append({"file":file,"cell":board.view.cell_size,"font":board.clue_font_size(),"status":status,"layout_valid":board.layout_valid,"glyph_risk":board.glyph_risk})
	var f := FileAccess.open(output.path_join("native-render.json"),FileAccess.WRITE)
	f.store_string(JSON.stringify({"renderer":RenderingServer.get_current_rendering_method(),"display":DisplayServer.get_name(),"definition":"candidate-definition.json","captures":metadata},"  ")+"\n")
	quit()
