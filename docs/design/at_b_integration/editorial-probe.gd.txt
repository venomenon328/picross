extends SceneTree
# One-off AT-B editorial inspection. Prepared isolated display states, no playthrough.
const Store = preload("res://model/save_store.gd")
var app
var surface: SubViewport
var output: String
var captures: Array = []

func _initialize() -> void:
	call_deferred("run")

func frames() -> void:
	for i in range(5): await process_frame
	await RenderingServer.frame_post_draw

func shot(name: String, ids: Array, target: Rect2 = Rect2()) -> void:
	await frames()
	var im: Image = surface.get_texture().get_image()
	assert(im.save_png(output.path_join(name+".png")) == OK)
	captures.append({"file": name+".png", "ids": ids, "ui_scale": app.ui_scale, "viewport": [surface.size.x, surface.size.y], "target_rect": [target.position.x,target.position.y,target.size.x,target.size.y]})

func run() -> void:
	output = OS.get_environment("AT_B_OUTPUT")
	Store.test_root_override = OS.get_environment("P1_TEST_SAVE_ROOT")
	assert(not output.is_empty() and not Store.test_root_override.is_empty())
	DirAccess.make_dir_recursive_absolute(Store.test_root_override)
	surface = SubViewport.new()
	surface.gui_embed_subwindows = true
	surface.size = Vector2i(1920,1080)
	surface.render_target_update_mode = SubViewport.UPDATE_ALWAYS
	root.add_child(surface)
	app = load("res://main.tscn").instantiate()
	surface.add_child(app)
	app.set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)
	await frames()
	assert(app.sessions.size() == 26 and app.catalog_error.is_empty())
	for s in app.sessions:
		var changes: Array = []
		for y in range(s.player.height):
			for x in range(s.player.width):
				var value: int = int(s.definition.solution[y][x])
				changes.append({"index": y*s.player.width+x, "before": -1, "after": value})
		assert(s.player.commit(changes) and s.is_solution())
		s.completed = true
	for u in [1.0,1.25]:
		app.set_ui_scale(u)
		app.show_album()
		await frames()
		# Scroll each actual card into view; filenames bind the inspected target.
		for s in app.sessions:
			var id: String = str(s.definition.id)
			app.album.choose_category(str(app.album.metadata[id].category_id))
			await frames()
			var index: int = app.album.visible_ids.find(id)
			assert(index >= 0)
			app.album.scroll.ensure_control_visible(app.album.grid.get_child(index))
			await frames()
			var target: Rect2 = app.album.choices[index].get_global_rect()
			assert(app.album.scroll.get_global_rect().encloses(target))
			await shot("catalog-"+id+"-ui"+str(roundi(u*100)), [id], target)
		for s in app.sessions:
			app.select_by_id(str(s.definition.id))
			await frames()
			assert(app.ending.visible and app.reveal_view.artwork != null)
			await shot("completion-"+str(s.definition.id)+"-ui"+str(roundi(u*100)), [s.definition.id])
		app.show_album()
	var f := FileAccess.open(output.path_join("captures.json"),FileAccess.WRITE)
	f.store_string(JSON.stringify({"captures":captures,"limitations":"Prepared isolated solved display states; no real saves, no solver, no playthrough, no lifecycle claim."},"\t")+"\n")
	f.close()
	print("AT_B_VISUAL_CAPTURE_OK ",captures.size())
	quit(0)
